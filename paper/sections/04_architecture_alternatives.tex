\begin{enumerate}
  \item \textbf{Stage 0 (Analytical MDL Header \& $\SL(C)$-\texttt{WSIL} Alignment):} Estimate $\hat{\phi} = (\phi_{\mathrm{acq}},\phi_{\mathrm{head}})$ in closed form (or trigger zero-rounding \texttt{Palette} mode when unique colors $|\mathcal{U}(X)| \le 512$ lowers MDL cost). Apply site-wise $\SL(C)$-\texttt{WSIL} bijection $y_0(u) = \mathcal{W}_{\phi_{\mathrm{acq}}}(x(u)) \in \Z^C$ so channel $c=0$ concentrates high-SNR luminance/green energy and $c \ge 1$ hold decorrelated opponent/ISP residuals.
  \item \textbf{Stage 1 (\texttt{RITC}-Optimal Shallow Quincunx Tower, $S=4$):} At each quincunx scale $s \in \{0,\dots,3\}$ ($\Lambda_{s+1} = Q\Lambda_s$), execute \emph{strictly one} anti-aliased 4-neighbor predict-update integer lift:
  \begin{equation}
    r_s(v) = y_s(v) - \Big\lfloor \frac{1}{4}\!\sum_{u \in \mathcal{N}_s^\diamond(v)}\! y_s(u) + \frac{1}{2}\Big\rfloor, \qquad
    y_{s+1}(u) = y_s(u) + \Big\lfloor \frac{1}{8}\!\sum_{v \in \mathcal{N}_s^\diamond(u)}\! r_s(v) + \frac{1}{2}\Big\rfloor.
    \label{eq:quincunx-lift}
  \end{equation}
  Partition $r_s$ into 3 causal spatial sub-cosets $r_s^{(1)}, r_s^{(2)}, r_s^{(3)}$ (isolating $1\%$ seed $x_{\mathrm{seed}} \subset r_0^{(3)}$) and code channels $c = 0 \to \dots \to C-1$ sequentially per parallel spatial coset.
  \item \textbf{Stage 2 (Integer Sheaf-Kalman + \texttt{I-SSM} Backbone):} Conditioned solely on causally decoded $(y_{>s}, r_s^{(<k)}, c'<c, \hat{\phi})$, evaluate in pure \texttt{INT8}/\texttt{INT16}/\texttt{INT32} arithmetic: bilateral sheaf bridge means $(\hat{\mu}_+, \hat{\mu}_-, \pi_{\mathrm{bridge}})$, $M=8$ $D_8$-equivariant Sheaf-Kalman matches, 4-way \texttt{Q15.16}/\texttt{Q1.30} \texttt{I-SSM} features $h_s(v) \in \Z^{64}$, and closed-form Ridge correction $\phi_{\mathrm{head}}^\top h_s(v)$.
  \item \textbf{Stage 3 (Fractional Bimodal Student-$t$ \& \texttt{Dyadic-Core66 rANS}):} Parameterize discrete residual $r \in \Z$ via sub-integer $\mu_\pm \in \Z[2^{-16}]$, $\sigma \in \Z[2^{-12}]$, and $\nu \in \{2.5, 3.5, 6.0, \infty\}$:
  \begin{equation}
    Q(r \mid \mathcal{C}) = \sum_{\iota \in \{+,-\}} \pi_\iota \Big[F_\nu\!\Big(\frac{r + 0.5 - \mu_\iota}{\sigma}\Big) - F_\nu\!\Big(\frac{r - 0.5 - \mu_\iota}{\sigma}\Big)\Big],
  \end{equation}
  evaluated via a normative $2\,\mathrm{KB}$ integer PCHIP LUT and streamed over 32-way SIMD \texttt{Dyadic-Core66 rANS} lanes.
\end{enumerate}

\subsection{Bijectivity, Minimax Rate Bound, and Parallel Complexity}

\begin{theorem}[Exact Integer Bijectivity and Cross-ISA Determinism]
\label{thm:bijectivity}
For every $X \in \Z_M^{H\times W\times C}$ and arbitrary x86, ARM, or GPU targets, \texttt{SLGF} satisfies $\mathrm{Dec}(\mathrm{Enc}(X)) \equiv X$ bit-identically.
\end{theorem}
\begin{proof}
Every \texttt{Dyadic-Core66 rANS} and \texttt{S\textsuperscript{3}-BB-rANS} transition uses strictly positive integer counts $f_k \ge 1$ with $\sum_{k=0}^{65} f_k = 2^{16}$, forming an exact algebraic bijection on $\N \times \{0,\dots,65\}$. All conditioning operators (\texttt{Q1.30} balanced-tree \texttt{I-SSM}, \texttt{INT8}/\texttt{INT32} convolutions, integer reciprocal Sheaf-Kalman weights, and integer PCHIP $F_\nu$) execute zero IEEE-754 operations on identical causal integer inputs, yielding identical $f_k$ across all ISAs. Finally, \eqref{eq:quincunx-lift} inverts exactly by subtracting $\lfloor \frac{1}{8}\sum r_s + \frac{1}{2}\rfloor$ before adding $\lfloor \frac{1}{4}\sum y_s + \frac{1}{2}\rfloor$, and $\SL(C)$-\texttt{WSIL} inverts by reversing its unitriangular integer shears (\Cref{sec:appendix-ops}).
\end{proof}

\begin{theorem}[Expected Codelength and Minimax Regret Bound]
\label{thm:rate-bound}
Under image law $P$ with acquisition manifold $\mathcal{M}_{\phi^*}$ and local shot noise $\sigma_\eta^2(u)$, expected \texttt{SLGF} rate in $\bpsp$ obeys
\begin{equation}
  \E[R_{\mathrm{SLGF}}(X)] \le \frac{H(X)}{HWC} + \underbrace{\frac{K_{\mathrm{eff}}\log_2(HWC)}{2HWC}}_{<\,0.002\text{ (MDL regret)}} + \underbrace{\delta_{\mathrm{RITC}}^{(L=1)}}_{\le\,0.021} + \delta_{\mathrm{KL}}^{\mathrm{sheaf+SK+SSM}}(P \| Q_\Theta) + \underbrace{\delta_{\mathrm{rANS}}}_{\le\,0.0013}.
\end{equation}
\end{theorem}

\noindent\textbf{Complexity Profile:} Decoding executes $S=4$ scales $\times$ $3$ sub-cosets $\times$ $C=3$ channels $=$ \textbf{$36$ resolution-independent parallel tensor launches} (plus $4\times 4$ wavefronts on base $a_4$ of size $\frac{1}{16}HW$). \texttt{SLGF-Fast} requires $\approx 18\text{K}$ \texttt{INT8/INT16} MACs/px ($\approx 8\,\mathrm{ms}$ GPU / $\approx 42\,\mathrm{ms}$ 8-core SIMD CPU decode per MP; $25\text{--}65\,\mathrm{ms}$ encode), with $512\times 512$ haloed tiling capping peak memory below $24\,\mathrm{MB}$ even on 8K/16-bit volumes.
