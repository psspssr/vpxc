\section{Six Frontier Mathematical Frameworks on \texorpdfstring{$\Z^d$}{Z\textasciicircum d}}
\label{sec:frameworks}

\subsection{Whitehead--Symplectic Integer Lifting (\texttt{WSIL}) over \texorpdfstring{$\SL(C, \R)$}{SL(C, R)}}
Unitriangular integer shears~\citep{sweldens1998,hoogeboom2019idf} (\texttt{YCoCg-R}, \texttt{SubtractGreen}) cannot execute continuous channel rescalings or arbitrary volume-preserving eigenframe rotations on $\Z^C$. By Whitehead's lemma in algebraic K-theory, any hyperbolic squeeze $\operatorname{diag}(s, s^{-1}) \in \SL(2,\R)$ ($s>0$) factors over elementary shears $U(\alpha) = \begin{psmallmatrix}1 & \alpha\\0 & 1\end{psmallmatrix}$ and $L(\beta) = \begin{psmallmatrix}1 & 0\\\beta & 1\end{psmallmatrix}$ for gauge $b \neq 0$ as $\operatorname{diag}(s,s^{-1}) = U(\frac{s-1}{b})L(b)U(\frac{s^{-1}-1}{b})L(-bs)$. Setting $b=1$ yields a 4-step bijection on $\Z^2$ with \textbf{only 3 rounding operations} (Step~3 is exact in $\Z$) and \textbf{identically zero rounding variance at $s = 1$}:
\begin{equation}
  u_2 = x_2 - \round{s x_1}, \quad
  u_1 = x_1 + \round{(s^{-1}-1)u_2}, \quad
  y_2 = u_2 + u_1, \quad
  y_1 = u_1 + \round{(s-1)y_2}.
  \label{eq:wsil-2d}
\end{equation}
Verified across $5\times 10^4$ uniform samples on $[-1000,1000]^2$, \eqref{eq:wsil-2d} has $0$ inversion errors and $\Var(y_1 - sx_1) = 0.0000$ at $s=1.00$ ($0.2038$ at $s=1.25$; $0.2500$ at $s=2.00$). For arbitrary $W \in \SL(C,\R)$ ($C=3$ RGB or $C=4$ Bayer $G_1 R B G_2$), pivoted factorization $PW = L\operatorname{diag}(d_1,\dots,d_C)U$ ($\prod_c d_c = 1$) decomposes the diagonal into $C-1$ nested 2D \texttt{WSIL} squeezes flanked by unitriangular integer shears. On rotated camera CCMs ($N=10^5$, latent stds $(35.0, 4.5, 1.8)$), $\SL(3)$-\texttt{WSIL} achieves $4.8440\bpsp$ vs.\ $5.4495\bpsp$ for \texttt{YCoCg-R} (\textbf{$-0.6055\bpsp$ gain}, within $0.073\bpsp$ of the $4.7709\bpsp$ unquantized latent floor).

\subsection{Multiscale Quincunx Lattice Towers and Cellular Sheaf Line-Processes}
Replacing separable $2\times 2$ dyadic Haar/\texttt{Squeeze} splits (which alias $45^\circ$ contours) with the commensurable $\sqrt{2}$-dilation \textbf{quincunx filtration} $\Z^2 = \Lambda_0 \supset \Lambda_1 \supset \dots \supset \Lambda_S$ generated by $Q = \begin{psmallmatrix}1 & 1\\1 & -1\end{psmallmatrix}$ ($\det Q = -2, Q^2 = 2I$) alternates axial and diagonal diamond stencils isotropically and isolates native Bayer Green sites onto $\Lambda_1$. Equipping scale graph $(V_s, E_s)$ with a \textbf{discrete cellular sheaf} $\mathcal{F}_s$---stalks $\mathcal{F}_s(u)=\R^C$, affine connections $\rho_{u\to v} \in \Aff(C)$, and edge-stopping conductances $\omega_{uv} = (1 + \|\rho_{u\to v} a_s(u) - a_s(v)\|_2^2/\kappa_s^2)^{-1}$---yields harmonic section interpolation $\hat{\mu}_{\mathrm{sheaf}}(v) = (\sum_u \omega_{uv})^{-1}\sum_u \omega_{uv}\rho_{u\to v}a_s(u)$. Across discontinuities ($\|a_s(u^+) - a_s(u^-)\| \gg \sigma_{\mathrm{sm}}$), instead of blurring across the jump, the predictor emits an explicit \textbf{2-component bilateral bridge mixture} over $a_s(u^+)$ and $a_s(u^-)$.

\subsection{Gauge-Equivariant Patch Bundles and Sheaf-Kalman Optimal Transport}
Modeling image patches on the semidirect fiber bundle $\mathcal{E} = \Z^2 \ltimes (D_8 \times \Aff_1(\R))$ mods out dihedral symmetries $g_m \in D_8$ and affine illumination shifts $(\alpha_m,\beta_m)$. Given $M$ gauge-aligned non-local matches $\tilde{y}_m(u) = s(u) + \tilde{\eta}_m$ ($\Var(\tilde{\eta}_m) = \sigma_\eta^2$) and local sheaf prior $(\hat{\mu}_{\mathrm{loc}}, \sigma_{\mathrm{loc}}^2)$, \textbf{precision-weighted Sheaf-Kalman fusion} yields the exact MMSE / minimum-entropy Gaussian-Student parameters:
\begin{equation}
  \hat{\mu}_{\mathrm{SK}}(u) = \frac{\sigma_{\mathrm{loc}}^{-2}\hat{\mu}_{\mathrm{loc}}(u) + \sigma_\eta^{-2}\sum_{m=1}^M \tilde{y}_m(u)}{\sigma_{\mathrm{loc}}^{-2} + M\sigma_\eta^{-2}}, \qquad
  \sigma_{\mathrm{pred}}^2(u) = \sigma_\eta^2 + \frac{1}{\sigma_{\mathrm{loc}}^{-2} + M\sigma_\eta^{-2}}.
  \label{eq:sheaf-kalman}
\end{equation}
For $\sigma_\eta^2 = 4.0$ and $\sigma_{\mathrm{loc}}^2 = 16.0$, hard patch subtraction (\texttt{IntraBC}/\texttt{Patches}) yields variance $2\sigma_\eta^2 = 8.0$ ($+0.5000\bpsp$ above $h_\eta$), whereas \eqref{eq:sheaf-kalman} saves \textbf{$-1.0085\bpsp$} ($M=4$) and \textbf{$-1.0784\bpsp$} ($M=8$) over local prediction (\textbf{$-0.4175\bpsp$} over patch subtraction, just $+0.0825\bpsp$ above the quantum floor), while continuously collapsing to $0\bpsp$ exact $D_8$ block copy as $\sigma_\eta \to 0$.

\subsection{Rissanen Two-Part Analytical Minimum Description Length (\texttt{MDL}) Priors}
Rather than running $8.45\,\mathrm{s}$ of test-time SGD, we minimize Rissanen's two-part MDL objective~\citep{rissanen1984} $\hat{\phi} = \arg\min_{\phi \in \Phi_Q}\{L_{\mathrm{hdr}}(\phi) - \log_2 P(X\mid \phi)\}$ \textbf{analytically in $<10\,\mathrm{ms}$} via two headers with minimax regret $\frac{K_{\mathrm{eff}}}{2N}\log_2 N$:
\begin{enumerate}
  \item \textbf{Acquisition Header $\phi_{\mathrm{acq}}$ ($\approx 1{,}154\,\mathrm{bits} \approx 0.00098\bpsp$ Kodak):} 2-bit CFA/4:2:0 phase $p_{\mathrm{CFA}} \in \Z_2^2$, 8-knot monotone PCHIP inverse tone curve ($96\,\mathrm{bits}$), 8-DOF $\SL(3)$-\texttt{WSIL} multipliers ($96\,\mathrm{bits}$), and $4\times 24$ \texttt{Q4.6} parity-conditioned demosaic taps ($960\,\mathrm{bits}$), gated by exact check $\Delta L_{\mathrm{MDL}} < 0$.
  \item \textbf{Closed-Form Ridge Head $\phi_{\mathrm{head}}$ ($\approx 557\,\mathrm{bits} \approx 0.00047\bpsp$ Kodak):} Single-shot $64\times 64$ Cholesky solve $\hat{w} = (\Phi^\top\Phi + \lambda_{\mathrm{MDL}}I)^{-1}\Phi^\top r_0$ quantized to \texttt{Q3.5} and Laplace-coded, delivering \textbf{$-0.3528\,\mathrm{bits/px}$} net gain on shifted sensor statistics ($N_{\mathrm{px}}=393{,}216$) in $<3\,\mathrm{ms}$ CPU time.
\end{enumerate}

\subsection{Canonical Fixed-Point Ring State-Space Mixing (\texttt{I-SSM}) \& \texttt{Dyadic-Core66 rANS}}
\texttt{I-SSM} replaces non-deterministic Softmax attention with a normative integer scan along 4 lattice directions on each decoded coarse coset: intra-chunk ($B=16$) \texttt{Q15.16} recurrence $h_t = ((A_t \odot h_{t-1} + 2^{15})\gg 16) + B_t x_t$ ($A_t \in [0, 2^{16}-1]$) bridged across chunks by a \textbf{canonical balanced binary-tree \texttt{Q1.30} prefix scan}, matching sequential \texttt{Q16} within $\le 1\;\mathrm{LSB}$ ($1.5\times 10^{-5}$) in $O(B + \log_2(N/B))$ parallel depth with $100\%$ cross-ISA bit-identity.

To eliminate single-image bits-back cold-start waste~\citep{townsend2019bbans,kingma2019bitswap}, \textbf{Tail-Seeded Split-Lattice Bits-Back (\texttt{S\textsuperscript{3}-BB-rANS})} encodes an isolated $1\%$ fine-lattice seed $x_{\mathrm{seed}} \subset r_0^{(3)}$ under $Q(\cdot\mid x_{\mathrm{main}},\hat{\phi})$ into $\mathcal{S}_0$, pops tile latents $z \sim q_\Psi(\cdot\mid x_{\mathrm{main}})$, and pushes $x_{\mathrm{main}}$ and $z$, achieving \textbf{$0\,\mathrm{bits}$ auxiliary seed waste}. Finally, \textbf{\texttt{Dyadic-Core66 rANS}} shifts residuals by $b(\sigma) = \max(0, \lceil \log_2(\sigma/2.5)\rceil)$ into $65$ core quotient bins $[-32,32]$ plus $1$ escape bin ($M=2^{16}$, $132$-byte L1 table), holding quantization + tail overhead below $0.0013\,\mathrm{bits/sample}$ across 8--16-bit scales (\Cref{tab:rans-audit}, vs.\ $+0.0104\,\mathrm{bits}$ for dense $K=513$ tables).
